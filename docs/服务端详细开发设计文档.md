## 4. LangGraph 主工作流（采集闭环编排）

### 4.1 状态定义

```python
# graphify/state.py
from typing import TypedDict, Annotated
from operator import add
from graphify.schemas.plan import CollectPlan
from graphify.schemas.graph import ExtractionResult

class CollectState(TypedDict):
    kb_id: str
    instruction: str                       # 用户自然语言指令
    plan: CollectPlan | None
    candidate_urls: list[str]              # 发现的候选 URL
    fresh_urls: list[str]                  # 去重后保留
    fetched: Annotated[list[dict], add]    # {url, html, text}
    extractions: Annotated[list[ExtractionResult], add]
    report: dict                           # 采集统计
    errors: Annotated[list[str], add]
```

### 4.2 主图结构


### 4.3 编排骨架

```python
# graphify/graph_workflow.py
from langgraph.graph import StateGraph, START, END
from langgraph.checkpoint.postgres import PostgresSaver   # 或 SqliteSaver
from graphify.state import CollectState
from graphify.agents import planner, collector, graph_builder

def _plan_node(state: CollectState) -> dict:
    plan = planner.parse_instruction(state["kb_id"], state["instruction"])
    return {"plan": plan}

def _discover_node(state: CollectState) -> dict:
    urls = collector.discover_urls(state["plan"])
    return {"candidate_urls": urls}

def _dedup_url_node(state: CollectState) -> dict:
    fresh = collector.filter_known_urls(
        state["plan"].kb_id, state["candidate_urls"], state["plan"].source_limit
    )
    return {"fresh_urls": fresh}

def _collect_node(state: CollectState) -> dict:
    fetched, errs = collector.run(state["plan"], state["fresh_urls"])
    return {"fetched": fetched, "errors": errs}

def _extract_node(state: CollectState) -> dict:
    results = [graph_builder.extract(item) for item in state["fetched"]]
    return {"extractions": results}

def _build_node(state: CollectState) -> dict:
    report = graph_builder.persist(state["plan"].kb_id, state["extractions"])
    return {"report": report}

def build_workflow(checkpointer=None):
    g = StateGraph(CollectState)
    g.add_node("plan", _plan_node)
    g.add_node("discover", _discover_node)
    g.add_node("dedup_url", _dedup_url_node)
    g.add_node("collect", _collect_node)
    g.add_node("extract_kg", _extract_node)
    g.add_node("build_graph", _build_node)

    g.add_edge(START, "plan")
    g.add_edge("plan", "discover")
    g.add_edge("discover", "dedup_url")
    g.add_edge("dedup_url", "collect")
    g.add_edge("collect", "extract_kg")
    g.add_edge("extract_kg", "build_graph")
    g.add_edge("build_graph", END)
    return g.compile(checkpointer=checkpointer)
```

> 持久化：使用 LangGraph Checkpointer，以 `thread_id=task_id` 实现断点续跑与幂等重入。节点函数保持**纯函数 + 注入依赖**风格，便于单测（见第 9 章）。

## 5. Agent 与工具实现设计

### 5.1 Planner —— 指令解析（结构化输出）

使用 LLM 的结构化输出能力直接产出 `CollectPlan`，确定性强、易测试。

```python
# graphify/agents/planner.py
from graphify.llm import get_llm
from graphify.schemas.plan import CollectPlan

PLANNER_SYS = """你是采集计划解析器。把用户的中文采集指令解析为结构化采集计划。
- 识别领域、主题、区域、来源数量(source_limit)、是否每日采集。
- 若提到"上下游/产业链"，结合行业知识扩展 upstream / downstream 词表。
只输出符合 schema 的结构化结果。"""

def parse_instruction(kb_id: str, instruction: str, llm=None) -> CollectPlan:
    llm = llm or get_llm()
    structured = llm.with_structured_output(CollectPlan)
    plan = structured.invoke(
        [("system", PLANNER_SYS), ("human", instruction)]
    )
    plan.kb_id = kb_id
    return plan
```

### 5.2 Collector —— deepagents 采集子 Agent

采集是典型「长流程 + 多工具 + 需规划」场景，用 `create_deep_agent` 承载；具体的发现/去重/抓取作为工具暴露给它。

```python
# graphify/agents/collector.py
from deepagents import create_deep_agent
from graphify.llm import model_id
from graphify.schemas.plan import CollectPlan
from graphify.tools.search import search_urls
from graphify.tools.fetch import fetch_page
from graphify.tools.dedup import check_duplicate
from graphify.tools.extract import extract_content
from graphify.store.source_repo import SourceRepo
from graphify.dedup.url_norm import normalize_url

COLLECTOR_SYS = """你是采集执行 Agent。给定采集计划与一批待抓取 URL：
1) 逐个调用 fetch_page 抓取；2) 调用 check_duplicate 做内容/语义去重；
3) 对非重复内容调用 extract_content 抽取正文与结构化字段；
跳过重复或抓取失败的 URL，最终返回成功抽取的内容列表。"""

_collector = create_deep_agent(
    model=model_id(),
    tools=[fetch_page, check_duplicate, extract_content],
    system_prompt=COLLECTOR_SYS,
)

# —— 确定性工具步骤直接用纯函数（便于测试、降低 LLM 成本）——
def discover_urls(plan: CollectPlan) -> list[str]:
    terms = plan.topics + plan.supply_chain.upstream + plan.supply_chain.downstream
    # 预留 1.5~2x 候选以抵消去重/失败损耗
    return search_urls.invoke({"terms": terms, "limit": int(plan.source_limit * 2),
                               "region": plan.region, "lang": plan.language})

def filter_known_urls(kb_id: str, urls: list[str], limit: int,
                      repo: SourceRepo | None = None) -> list[str]:
    repo = repo or SourceRepo()
    fresh = []
    for u in urls:
        nu = normalize_url(u)
        if not repo.url_seen(kb_id, nu):
            fresh.append(u)
        if len(fresh) >= limit:
            break
    return fresh

def run(plan: CollectPlan, urls: list[str]):
    """可选：交给 deep agent；或直接编排工具。默认走确定性编排以便测试。"""
    fetched, errors = [], []
    for u in urls:
        try:
            page = fetch_page.invoke({"url": u})
            dup = check_duplicate.invoke({"kb_id": plan.kb_id, "url": u,
                                          "text": page["text"]})
            if dup["is_duplicate"]:
                continue
            content = extract_content.invoke({"html": page["html"]})
            fetched.append({"url": u, **content})
        except Exception as e:           # 失败隔离，不影响整批
            errors.append(f"{u}: {e}")
    return fetched, errors
```

> 设计取舍：默认用确定性编排（`run`）保证可测与低成本；`_collector` deep agent 作为「复杂站点/需要自主规划」时的可切换路径。两者共享同一批工具。

### 5.4 Graph Builder (Graphify) —— 抽取 + 写图

```python
# graphify/agents/graph_builder.py
from graphify.llm import get_llm
from graphify.schemas.graph import ExtractionResult
from graphify.graphdb.dao import GraphDAO

SCHEMA_HINT = "允许的节点: Company,Industry,Component,Product,Metric,Person,NewsArticle; " \
              "允许的关系: BELONGS_TO,SUPPLIES,PART_OF_SUPPLY_CHAIN,PRODUCES,HAS_METRIC,MENTIONED_IN"

def extract(item: dict, llm=None) -> ExtractionResult:
    llm = llm or get_llm()
    structured = llm.with_structured_output(ExtractionResult)
    res: ExtractionResult = structured.invoke([
        ("system", f"严格按受控 schema 抽取知识图谱三元组。{SCHEMA_HINT}"),
        ("human", item["text"][:8000]),
    ])
    res.source_url = item["url"]
    return res

def persist(kb_id: str, extractions: list[ExtractionResult],
            dao: GraphDAO | None = None, min_conf: float = 0.6) -> dict:
    dao = dao or GraphDAO()
    added_nodes = added_rels = added_snaps = skipped = 0
    for ex in extractions:
        src = dao.merge_source(ex.source_url)
        for ent in ex.entities:
            if ent.confidence < min_conf:
                skipped += 1; continue
            node = dao.resolve_and_merge_entity(kb_id, ent)  # 消歧 + MERGE
            dao.link_source(node, src)
            added_nodes += 1
        for rel in ex.relations:
            if rel.confidence < min_conf:
                skipped += 1; continue
            dao.merge_relation(kb_id, rel); added_rels += 1
        for snap in ex.snapshots:
            dao.append_snapshot(kb_id, snap); added_snaps += 1   # 时序追加, 不覆盖
    return {"nodes": added_nodes, "relations": added_rels,
            "snapshots": added_snaps, "skipped_low_conf": skipped}
```

### 5.5 Query Agent —— NL2Cypher + 答案生成

```python
# graphify/agents/query_agent.py
from graphify.llm import get_llm
from graphify.schemas.query import QueryResult
from graphify.graphdb.dao import GraphDAO
from graphify.tools.cypher import is_read_only

NL2CYPHER_SYS = """把用户问题转为 Neo4j 只读 Cypher。
- 仅允许 MATCH/RETURN/WHERE/ORDER/LIMIT，禁止 CREATE/MERGE/DELETE/SET。
- 必须带 kb_id 过滤与 LIMIT。- 使用参数化占位符。"""

def answer(kb_id: str, question: str, dao: GraphDAO | None = None, llm=None) -> QueryResult:
    dao = dao or GraphDAO(); llm = llm or get_llm()
    cypher = llm.invoke([("system", NL2CYPHER_SYS),
                         ("human", f"kb_id={kb_id}; 问题: {question}")]).content
    if not is_read_only(cypher):                 # 安全护栏
        raise ValueError("生成的 Cypher 含写操作，已拦截")
    rows = dao.run_read(cypher, {"kb_id": kb_id})
    summary = llm.invoke([("system", "基于查询结果用中文简洁作答，并标注来源URL。"),
                          ("human", str(rows)[:6000])]).content
    sources = [r.get("url") for r in rows if isinstance(r, dict) and r.get("url")]
    return QueryResult(cypher=cypher, rows=rows, answer=summary, sources=sources)
```

## 6. 去重模块设计

### 6.1 URL 规范化

```python
# graphify/dedup/url_norm.py
from urllib.parse import urlsplit, urlunsplit, parse_qsl, urlencode

_DROP = {"utm_source","utm_medium","utm_campaign","utm_term","utm_content","ref","fbclid","gclid"}

def normalize_url(url: str) -> str:
    s = urlsplit(url.strip())
    scheme = (s.scheme or "https").lower()
    netloc = s.netloc.lower()
    query = urlencode(sorted((k, v) for k, v in parse_qsl(s.query) if k not in _DROP))
    path = s.path.rstrip("/") or "/"
    return urlunsplit((scheme, netloc, path, query, ""))   # 去 fragment
```

### 6.2 内容指纹（SimHash）

```python
# graphify/dedup/simhash.py
def simhash(text: str, bits: int = 64) -> int: ...
def hamming(a: int, b: int) -> int: return bin(a ^ b).count("1")
def is_near_dup(a: int, b: int, threshold: int = 3) -> bool:
    return hamming(a, b) <= threshold
```

### 6.3 三层去重逻辑（`check_duplicate` 内部）

```
1) URL级:  normalize_url → source_repo.url_seen(kb_id, nu)
            命中 → 检查 content_hash 是否变化:
                   - 未变 → is_duplicate=True
                   - 变了 → is_duplicate=False, changed=True (走时序更新)
2) 内容级:  simhash(text) 与近期指纹比对, hamming<=阈值 → 重复
3) 语义级:  embedding(text) 向量近邻, cos>=0.95 → 重复 (可选开关)
每层命中写 dedup_log 审计。
```

```python
        # 消歧: 名称/别名/ticker 匹配; 命中则复用, 否则新建
        q = """
        MERGE (n:%s {kb_id:$kb, name:$name})
        ON CREATE SET n += $props, n.created_at=datetime()
        ON MATCH  SET n += $props
        RETURN id(n) AS id
        """ % ent.label
        with self.driver.session() as s:
            return s.run(q, kb=kb_id, name=ent.name, props=ent.props).single()["id"]

    def merge_relation(self, kb_id: str, rel: Relation):
        q = """
        MATCH (a {kb_id:$kb, name:$src}), (b {kb_id:$kb, name:$dst})
        MERGE (a)-[r:%s]->(b)
        ON CREATE SET r += $props
        """ % rel.type
        with self.driver.session() as s:
            s.run(q, kb=kb_id, src=rel.source_name, dst=rel.target_name, props=rel.props)

    def append_snapshot(self, kb_id: str, snap: MetricSnapshot):
        q = """
        MERGE (c:Company {kb_id:$kb, name:$company})
        MERGE (m:Metric  {kb_id:$kb, name:$metric})
        MERGE (c)-[:HAS_METRIC]->(m)
        CREATE (m)-[:HAS_SNAPSHOT]->(:MetricSnapshot {value:$v, ts:date($ts), unit:$unit})
        """
        with self.driver.session() as s:
            s.run(q, kb=kb_id, company=snap.company, metric=snap.metric,
                  v=snap.value, ts=snap.ts, unit=snap.unit)

    def run_read(self, cypher: str, params: dict) -> list[dict]:
        with self.driver.session(default_access_mode="READ") as s:
            return [r.data() for r in s.run(cypher, **params)]

    def expand_node(self, kb_id: str, node_id: int, rel_types: list[str], limit=50):
        q = """MATCH (n)-[r]-(m) WHERE id(n)=$id AND type(r) IN $types
               RETURN n, r, m LIMIT $limit"""
        return self.run_read(q, {"id": node_id, "types": rel_types, "limit": limit})
```

## 9. 命令行接口（CLI，调试与测试用）

CLI 用 Typer 实现，逐节点可单独触发，方便研发逐步调试。

```python
# graphify/cli/main.py
import json, typer
from graphify.agents import planner, graph_builder, query_agent
from graphify.agents import collector
from graphify.graph_workflow import build_workflow
from langgraph.checkpoint.sqlite import SqliteSaver

app = typer.Typer(help="Graphify 服务端调试 CLI")

@app.command()
def plan(kb_id: str, instruction: str):
    """仅解析采集指令为 CollectPlan（不采集）。"""
    p = planner.parse_instruction(kb_id, instruction)
    typer.echo(p.model_dump_json(indent=2, ensure_ascii=False))

@app.command()
def discover(kb_id: str, instruction: str):
    """解析 + URL 发现，打印候选 URL。"""
    p = planner.parse_instruction(kb_id, instruction)
    urls = collector.discover_urls(p)
    typer.echo(json.dumps(urls, ensure_ascii=False, indent=2))

@app.command()
def fetch(url: str):
    """抓取单个 URL，打印正文摘要。"""
    from graphify.tools.fetch import fetch_page
    page = fetch_page.invoke({"url": url})
    typer.echo(page["text"][:1000])

@app.command()
def dedup(kb_id: str, url: str):
    """对单个 URL 跑三层去重，打印结果。"""
    from graphify.tools.fetch import fetch_page
    from graphify.tools.dedup import check_duplicate
    page = fetch_page.invoke({"url": url})
    typer.echo(json.dumps(check_duplicate.invoke(
        {"kb_id": kb_id, "url": url, "text": page["text"]}), ensure_ascii=False, indent=2))

@app.command()
def collect(kb_id: str, instruction: str, dry_run: bool = False):
    """端到端运行采集工作流。"""
    wf = build_workflow(checkpointer=SqliteSaver.from_conn_string("checkpoints.db").__enter__())
    out = wf.invoke({"kb_id": kb_id, "instruction": instruction},
                    config={"configurable": {"thread_id": f"{kb_id}-cli"}})
    typer.echo(json.dumps(out["report"], ensure_ascii=False, indent=2))

@app.command()
def ask(kb_id: str, question: str):
    """自然语言查询知识图谱。"""
    res = query_agent.answer(kb_id, question)
    typer.echo(f"Cypher: {res.cypher}\n\n答案: {res.answer}\n来源: {res.sources}")

@app.command()
def expand(kb_id: str, node_id: int, rel_types: str = "SUPPLIES,HAS_METRIC"):
    """图形化探索：展开某节点的邻居（命令行模拟点击）。"""
    from graphify.graphdb.dao import GraphDAO
    rows = GraphDAO().expand_node(kb_id, node_id, rel_types.split(","))
    typer.echo(json.dumps(rows, ensure_ascii=False, indent=2, default=str))

if __name__ == "__main__":
    app()
```

## 12. 设计要点小结

1. **混合编排**：确定性步骤用纯函数 LangGraph 节点（可测、低成本），需自主规划的采集用 deepagents 子 Agent，二者共享工具。
2. **幂等与续跑**：LangGraph Checkpointer + Neo4j `MERGE`，重复执行不产生重复数据。
3. **去重三层 + 变化检测**：保证"每日不重复"，同时让"会变化的数据"以时序快照形式持续累积。
4. **安全查询**：NL2Cypher 只读护栏 + 参数化 + kb_id 隔离。
5. **可测性优先**：依赖注入（llm/dao/repo 均可传入替身），单测覆盖去重、安全、编排、持久化核心路径，集成测试用 testcontainers 验证 Neo4j 真实行为。
